اولین سرنخ توی فایل اول این بود که تعداد کاراکتر "\" با تعداد کاراکتر "8" برابر بود.
و از اون طرف تعداد کارکتر "]" نیز با تعداد کاراکتر "9" در فایل اصلی برابر بود.
بعد از بررسی های مختلف این شباهت ها فهمیدم در سیستم utf-8 اگر decimal value رو منهای 35 کنیم. این کاراکتر ها به هم میرسند.

```
"[" - 35 = "9"
"\" - 35 = "8"
```

```python
import base64
import os

# Constants
subtraction_value = 0
is_value_set = False
NEWLINE_REPLACEMENT_FOR_WINDOWS_ONLY = "\r\n"

file = r"Z:\3.txt"

# Step 1: decode base64
with open(file, 'r', encoding='utf-8') as f:
    base64_source_file = f.read().replace("_", "/").replace("-", "+")

# Add padding if needed
mod4 = len(base64_source_file) % 4
if mod4 > 0:
    base64_source_file += '=' * (4 - mod4)

# Decode base64
try:
    decoded_bytes = base64.b64decode(base64_source_file)
    text = decoded_bytes.decode('utf-8')
except Exception as e:
    print(f"Error decoding base64: {e}")
    exit(1)

# Step 2: subtract all utf-8 chars by x value
# x = first char value - 100
result_list = []

for i in range(len(text)):
    char = text[i]
    a = ord(char)
    
    if not is_value_set:
        subtraction_value = a - 100
        is_value_set = True
    
    # Calculate new character code
    new_code = a if a < subtraction_value else a - subtraction_value
    
    # Safety check
    if a < subtraction_value:
        print(f"this character is below {subtraction_value} : {char},{a}")
        continue
    
    # Convert back to character
    b = chr(new_code)
    
    
    result_list.append(b)

# Step 3: after this step the content is obfuscated
# for example "<div" is look like "d<vi"
# and we should swap chars two by two
# you can see it by removing this part
for j in range(1, len(result_list), 2):
    a = result_list[j]
    b = result_list[j - 1]
    
    result_list[j] = b
    result_list[j - 1] = a

# Step 4: combine everything and show the result
with open("output.txt", "w", encoding="utf-8") as f:
    f.write("".join(result_list))

print("we are done!!!")

```

// Constants

//x = first char value - 100
var list = new List<string>();

for (int i = 0; i < text.Length; i++)

    if (!isValueSet)
    {
        subtractionValue = a - 100;
        isValueSet = true;
    }

    var b = char.ConvertFromUtf32(a < subtractionValue ? a : a - subtractionValue);


    //safety
    if (a < subtractionValue)
    {

    }

    //windows only
    if (b == "\n")
    {

    list.Add(b);
